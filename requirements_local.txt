# ========================================

# server and the React/Vite frontend locally for *viewing* existing runs and
# benchmarks. It intentionally excludes torch, vLLM, transformers, CUDA drivers,
# sentence-transformers, and the other heavy model-training packages that are
# required on the GPU HPC nodes.
#
# Install:
#   pip install -r requirements_local.txt
#   cd ui && npm install
#
# Run backend (without loading models):
#   AUTORED_LOAD_MODELS=0 python -m uvicorn server.main:app --host 127.0.0.1 --port 8000
#
# Run frontend dev server:
#   cd ui && npm run dev
